barrel() { # $1 = barrel path, $2 = original copy
  local dir; dir=$(dirname "$1")
  local out=""
  while IFS= read -r line; do
    if [[ "$line" =~ from\ \'\./([^\']+)\' ]]; then
      local mod="$dir/${BASH_REMATCH[1]}"
      if git ls-files --error-unmatch "$mod.ts" "$mod.tsx" >/dev/null 2>&1 \
